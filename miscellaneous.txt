# All types of graphs which are feasible using dash-polotly.


import dash
from dash import html, dcc,_dash_renderer
import dash_mantine_components as dmc
import plotly.express as px
import pandas as pd

_dash_renderer._set_react_version("18.2.0")

# Sample data for demonstration
df = px.data.iris()
df_bar = px.data.tips()
df_geo = px.data.gapminder()
df_funnel = pd.DataFrame({
    "stage": ["Awareness", "Interest", "Decision", "Action"],
    "count": [300, 200, 150, 100]
})

# Initialize the Dash app
app = dash.Dash(__name__)

# Define the app layout
app.layout = dmc.MantineProvider(
    children=dmc.Container([
        dmc.Title("Plotly Figure Examples", order=1),
        dmc.Space(h=20),

        dcc.Graph(
            id="scatter-plot",
            figure=px.scatter(
                df, x="sepal_width", y="sepal_length", color="species",
                title="Scatter Plot: Sepal Width vs Sepal Length"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="bubble-chart",
            figure=px.scatter(
                df, x="sepal_width", y="sepal_length", size="petal_length", color="species",
                title="Bubble Chart: Sepal Dimensions with Petal Length"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="line-plot",
            figure=px.line(
                df_geo[df_geo['country'] == 'India'], x="year", y="lifeExp",
                title="Line Plot: Life Expectancy in India Over Time"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="bar-plot",
            figure=px.bar(
                df_bar, x="day", y="total_bill", color="sex", barmode="group",
                title="Bar Chart: Total Bill by Day and Gender"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="pie-chart",
            figure=px.pie(
                df, names="species", title="Pie Chart: Distribution of Iris Species"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="box-plot",
            figure=px.box(
                df, x="species", y="petal_length", color="species",
                title="Box Plot: Petal Length Distribution by Species"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="violin-plot",
            figure=px.violin(
                df, x="species", y="sepal_width", color="species",
                title="Violin Plot: Sepal Width Distribution by Species"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="strip-plot",
            figure=px.strip(
                df, x="species", y="sepal_length", color="species",
                title="Strip Plot: Sepal Length by Species"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="density-heatmap",
            figure=px.density_heatmap(
                df_bar, x="total_bill", y="tip", title="Density Heatmap: Total Bill vs Tip"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="density-contour",
            figure=px.density_contour(
                df, x="sepal_width", y="sepal_length", color="species",
                title="Density Contour: Sepal Dimensions"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="sunburst",
            figure=px.sunburst(
                df_geo, path=["continent", "country"], values="pop",
                title="Sunburst Chart: Population by Continent and Country"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="treemap",
            figure=px.treemap(
                df_geo, path=["continent", "country"], values="gdpPercap",
                title="Treemap: GDP per Capita by Continent and Country"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="scatter-matrix",
            figure=px.scatter_matrix(
                df, dimensions=["sepal_width", "sepal_length", "petal_length", "petal_width"], color="species",
                title="Scatter Matrix: Iris Dataset"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="parallel-coordinates",
            figure=px.parallel_coordinates(
                df, dimensions=["sepal_width", "sepal_length", "petal_length", "petal_width"], color="species_id",
                title="Parallel Coordinates: Iris Dataset"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="parallel-categories",
            figure=px.parallel_categories(
                df_bar, dimensions=["sex", "day", "time"], color="total_bill",
                title="Parallel Categories: Tips Dataset"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="area-chart",
            figure=px.area(
                df_geo[df_geo['continent'] == 'Asia'], x="year", y="pop", color="country",
                title="Area Chart: Population of Asian Countries Over Time"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="funnel-chart",
            figure=px.funnel(
                df_funnel, x="stage", y="count",
                title="Funnel Chart: Marketing Stages"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="funnel-area-chart",
            figure=px.funnel_area(
                df_funnel, values="count", names="stage",
                title="Funnel Area Chart: Marketing Stages"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="scatter-geo",
            figure=px.scatter_geo(
                df_geo, locations="iso_alpha", color="continent", size="pop", projection="natural earth",
                title="Scatter Geo: Population by Country"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="choropleth-map",
            figure=px.choropleth(
                df_geo, locations="iso_alpha", color="lifeExp", hover_name="country",
                title="Choropleth Map: Life Expectancy by Country"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="scatter-3d",
            figure=px.scatter_3d(
                df, x="sepal_width", y="sepal_length", z="petal_length", color="species",
                title="3D Scatter Plot: Sepal and Petal Dimensions"
            )
        ),

        dmc.Space(h=20),

        dcc.Graph(
            id="line-3d",
            figure=px.line_3d(
                df_geo[df_geo['country'] == 'India'], x="year", y="pop", z="gdpPercap",
                title="3D Line Plot: India's Population and GDP Over Time"
            )
        ),

    ])
)

# Run the app
if __name__ == "__main__":
    app.run_server(debug=True)
